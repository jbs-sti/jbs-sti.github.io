% Police : Nunito, fichier variable du dépôt (axe de graisse). Le dossier
% du projet vient de l'environnement de Quarto ; à défaut, le dossier
% courant. Pas d'italique propre au fichier : inclinaison simulée.
\usepackage{fontspec}
\newcommand{\nunitodossier}{\directlua{tex.sprint(os.getenv("QUARTO_PROJECT_DIR") or ".")}/_extensions/vintage/fonts/}
\IfFileExists{\nunitodossier Nunito-VF.ttf}{%
\setmainfont{Nunito-VF}[Path=\nunitodossier, Extension=.ttf,
  UprightFeatures={RawFeature={axis={wght=400}}},
  BoldFont=*, BoldFeatures={RawFeature={axis={wght=800}}},
  ItalicFont=*, ItalicFeatures={RawFeature={axis={wght=400}}, FakeSlant=0.2},
  BoldItalicFont=*, BoldItalicFeatures={RawFeature={axis={wght=800}}, FakeSlant=0.2}]
\newfontfamily\nunitonoir{Nunito-VF}[Path=\nunitodossier, Extension=.ttf,
  RawFeature={axis={wght=1000}}]
}{\newcommand{\nunitonoir}{\bfseries}%
